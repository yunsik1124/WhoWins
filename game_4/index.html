// =================================================================== // ⚔️ [독립 결계 수립] 스파인과 이미지 Y값 상호 간섭 완전 차단 초식 // =================================================================== function renderHeroGraphic(containerId, heroId, isIngame = false) { const container = document.getElementById(containerId); if (!container) return; container.innerHTML = ''; container.style.position = 'relative'; // 인게임 상자 결계 해제 스펙 보존 if (isIngame) { container.style.width = '100%'; container.style.height = '100%'; container.style.overflow = 'visible'; } // ⚠️ 부모 상자의 변환을 초기화하여 자식들에게 이중 중첩되는 현상을 원천 차단하오! container.style.transform = 'none'; const basePath = `./assets/spine/hero_${heroId}`; const imgPath = `./assets/portraits/hero_${heroId}.png`; // ⚔️ 대협 전용 제어 혈 자리 (절대 보존) const INGAME_SPINE = { scale: 1.8, x: 1300, y: 950 }; const INGAME_IMAGE = { scale: 1.8, x: 1300, y: -950 }; const LOBBY_SPINE = { scale: 1.4, x: 0, y: 80 }; // ➔ 이제 오직 스파인에만 깨끗하게 먹히오! const LOBBY_IMAGE = { scale: 1, x: 0, y: 50 }; // ➔ 이제 오직 이미지에만 깨끗하게 먹히오! // [1] Fallback Image 객체 생성 및 배치 const fallbackImg = document.createElement('img'); fallbackImg.src = imgPath; fallbackImg.style.position = 'absolute'; fallbackImg.style.pointerEvents = 'none'; fallbackImg.style.zIndex = '50'; fallbackImg.style.display = 'none'; fallbackImg.style.objectFit = 'contain'; if (isIngame) { fallbackImg.style.bottom = '0px'; fallbackImg.style.left = '50%'; fallbackImg.style.width = '100%'; fallbackImg.style.height = '100%'; fallbackImg.style.transform = `translate(calc(-50% + ${INGAME_IMAGE.x}px), ${-INGAME_IMAGE.y}px) scale(${INGAME_IMAGE.scale})`; fallbackImg.style.transformOrigin = 'bottom center'; const completeMask = 'linear-gradient(to bottom, black 150px, transparent 150px)'; fallbackImg.style.maskImage = completeMask; fallbackImg.style.webkitMaskImage = completeMask; } else { fallbackImg.style.top = '0'; fallbackImg.style.left = '0'; fallbackImg.style.width = '100%'; fallbackImg.style.height = '100%'; // 부모의 간섭 없이 오직 자기 자신만 50px 제어하오 fallbackImg.style.transform = `translate(${LOBBY_IMAGE.x}px, ${LOBBY_IMAGE.y}px) scale(${LOBBY_IMAGE.scale})`; fallbackImg.style.transformOrigin = 'center center'; } container.appendChild(fallbackImg); // [2] Fetch Spine JSON 구동 fetch(`${basePath}.json`) .then(res => { if (!res.ok) throw new Error("스파인 무공 비급 호출 실패!"); return res.text(); }) .then(text => { // Spine 로드 성공 시 이미지는 철저히 숨기오 fallbackImg.style.display = 'none'; const spineDiv = document.createElement('div'); spineDiv.style.position = 'absolute'; // ✨ [화질 극대화] 늘어났을 때 흐려짐 방지 및 선명도 극대화 spineDiv.style.imageRendering = '-webkit-optimize-contrast'; spineDiv.style.imageRendering = 'crisp-edges'; container.appendChild(spineDiv); if (isIngame) { spineDiv.style.bottom = '0px'; spineDiv.style.left = '50%'; spineDiv.style.width = '100%'; spineDiv.style.height = '100%'; spineDiv.style.transform = `translate(calc(-50% + ${INGAME_SPINE.x}px), ${-INGAME_SPINE.y}px) scale(${INGAME_SPINE.scale})`; spineDiv.style.transformOrigin = 'bottom center'; const completeMask = 'linear-gradient(to bottom, black 150px, transparent 150px)'; spineDiv.style.maskImage = completeMask; spineDiv.style.webkitMaskImage = completeMask; } else { // ✨ 로비일 때 플레이어 왜곡 방지 및 독립적인 수치 제어 이식! // 외부 플레이어 정렬을 흐트러뜨리지 않고 spineDiv 자체에 완벽하게 박아버리오. spineDiv.style.top = '0'; spineDiv.style.left = '0'; spineDiv.style.width = '100%'; spineDiv.style.height = '100%'; spineDiv.style.transform = `translate(${LOBBY_SPINE.x}px, ${LOBBY_SPINE.y}px) scale(${LOBBY_SPINE.scale})`; spineDiv.style.transformOrigin = 'center center'; } // SpinePlayer 구동 및 최고 화질 세팅 조율 new spine.SpinePlayer(spineDiv, { jsonUrl: `${basePath}.json`, atlasUrl: `${basePath}.atlas`, animation: "idle", alpha: true, backgroundColor: "#00000000", showControls: false, preserveAspectRatio: true, viewport: { padLeft: "0%", padRight: "0%", padTop: "0%", padBottom: "0%" }, success: (player) => { setTimeout(() => { if (player) player.resize(); }, 50); }, error: (player, err) => { console.error("Spine 구동 실패, 이미지로 대체하오:", err); spineDiv.remove(); if (isIngame) { container.style.width = '420px'; container.style.height = '180px'; } fallbackImg.style.display = 'block'; } }); }) .catch(e => { console.error("리소스 패치 실패:", e); if (isIngame) { container.style.width = '420px'; container.style.height = '180px'; } fallbackImg.style.display = 'block'; }); }
